   ```python
   """Pre-computed indicator cache. Single-pass O(N) calculation, then per-bar O(1) lookup."""
   import pandas as pd
   import numpy as np


   def precompute_indicators(df_5m: pd.DataFrame) -> pd.DataFrame:
       """
       Add indicator columns to df_5m IN-PLACE-EQUIVALENT (returns same df with new cols).
       Computes: EMA20, ATR14, Highest20_shift1, Lowest20_shift1, plus any other recurring rollings.
       All computed once over full series — pandas handles this in C internally.
       """
       df = df_5m  # not copying — caller owns lifecycle
       if 'EMA20' not in df.columns:
           df['EMA20'] = df['close'].ewm(span=20, adjust=False).mean()
       if 'ATR14' not in df.columns:
           h, l, c = df['high'], df['low'], df['close'].shift(1)
           tr = pd.concat([h - l, (h - c).abs(), (l - c).abs()], axis=1).max(axis=1)
           df['ATR14'] = tr.rolling(14).mean()
       if 'Highest20_s1' not in df.columns:
           df['Highest20_s1'] = df['high'].rolling(20).max().shift(1)
       if 'Lowest20_s1' not in df.columns:
           df['Lowest20_s1'] = df['low'].rolling(20).min().shift(1)
       return df
   ```
